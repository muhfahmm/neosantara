# Jalankan dari root repository.
# Instal dependensi sekali saat setup awal:
npm install

# --- TAURI (DESKTOP) ---
# Menjalankan aplikasi Tauri bersama Next.js di http://localhost:3000:
npm run tauri:dev

# --- ELECTRON (DESKTOP) ---
# Menyalakan Next.js, menunggu server siap, lalu membuka Electron:
npm run electron:start

# Alias yang setara:
npm run electron:dev
